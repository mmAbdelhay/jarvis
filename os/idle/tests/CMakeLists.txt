file(GLOB JARVIS_IDLE_UNIT_TESTS CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/unit/tst_*.cpp)
list(FILTER JARVIS_IDLE_UNIT_TESTS EXCLUDE REGEX "tst_logindwatcher\\.cpp$")
foreach(source IN LISTS JARVIS_IDLE_UNIT_TESTS)
  get_filename_component(name ${source} NAME_WE)
  qt_add_executable(${name} ${source})
  target_link_libraries(${name} PRIVATE jarvis_idle_core Qt6::Test)
  target_compile_definitions(${name} PRIVATE JARVIS_IDLE_TEST_DATA="${CMAKE_CURRENT_SOURCE_DIR}/data")
  add_test(NAME ${name} COMMAND ${name})
  set_tests_properties(${name} PROPERTIES LABELS unit TIMEOUT 60)
endforeach()
if(JARVIS_IDLE_LINUX)
  find_program(DBUS_RUN_SESSION dbus-run-session)
  if(DBUS_RUN_SESSION)
    qt_add_executable(tst_logindwatcher unit/tst_logindwatcher.cpp support/FakeLogind.h support/FakeLogind.cpp)
    target_include_directories(tst_logindwatcher PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/support)
    target_link_libraries(tst_logindwatcher PRIVATE jarvis_idle_linux Qt6::Test)
    add_test(NAME tst_logindwatcher COMMAND ${DBUS_RUN_SESSION} -- $<TARGET_FILE:tst_logindwatcher>)
    set_tests_properties(tst_logindwatcher PROPERTIES LABELS dbus TIMEOUT 60)
  else()
    message(WARNING "dbus-run-session not found: tst_logindwatcher is not built (install dbus-daemon)")
  endif()
endif()
